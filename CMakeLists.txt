cmake_minimum_required(VERSION 3.20)
project(DriveTempMon VERSION 1.0.0 LANGUAGES CXX)

set(DTM_VERSION "${PROJECT_VERSION}" CACHE STRING "Release version (major.minor.patch)")
if(NOT DTM_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "DTM_VERSION must be major.minor.patch")
endif()
string(REPLACE "." "," DTM_VERSION_COMMA "${DTM_VERSION}")
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(dtm_policy STATIC src/policy.cpp)
target_include_directories(dtm_policy PUBLIC src)

include(CTest)
if(BUILD_TESTING)
  add_executable(policy_test tests/policy_test.cpp)
  target_link_libraries(policy_test PRIVATE dtm_policy)
  add_test(NAME temperature_policy COMMAND policy_test)
  add_executable(sensors_test tests/sensors_test.cpp)
  target_include_directories(sensors_test PRIVATE src)
  add_test(NAME sensor_parsers COMMAND sensors_test)
  foreach(test_target policy_test sensors_test)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /W4 /utf-8 /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    endif()
  endforeach()
endif()

if(WIN32)
  enable_language(RC)
  if(BUILD_TESTING)
    add_executable(config_test tests/config_test.cpp src/config.cpp src/startup.cpp)
    target_include_directories(config_test PRIVATE src)
    target_compile_definitions(config_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A000006)
    target_link_libraries(config_test PRIVATE dtm_policy shell32 ole32 oleaut32 uuid advapi32 taskschd)
    if(MSVC)
      target_compile_options(config_test PRIVATE /W4 /utf-8 /UNDEBUG)
    endif()
    add_test(NAME settings_persistence COMMAND config_test)
  endif()
  configure_file(src/app.rc.in generated/app.rc @ONLY)
  add_executable(DriveTempMon WIN32
    src/app.cpp src/config.cpp src/startup.cpp src/sensors.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc")
  target_include_directories(DriveTempMon PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(DriveTempMon PRIVATE
    UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
    _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A000006)
  target_link_libraries(DriveTempMon PRIVATE dtm_policy
    user32 gdi32 comctl32 shell32 shlwapi advapi32
    ole32 oleaut32 uuid sapi setupapi psapi taskschd)
  if(MSVC)
    target_compile_options(DriveTempMon PRIVATE /W4 /utf-8 /permissive- /Zc:__cplusplus /EHsc /guard:cf)
    target_compile_options(dtm_policy PRIVATE /W4 /utf-8 /permissive-)
    foreach(target DriveTempMon dtm_policy)
      target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL> $<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/Gw>)
    endforeach()
    target_link_options(DriveTempMon PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf
      $<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
  else()
    target_compile_options(DriveTempMon PRIVATE -Wall -Wextra -Os -ffunction-sections -fdata-sections)
    target_link_options(DriveTempMon PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--gc-sections)
  endif()
  install(TARGETS DriveTempMon RUNTIME DESTINATION .)
else()
  message(STATUS "Non-Windows host: building portable policy/parser tests only.")
endif()
